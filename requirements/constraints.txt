# Pin the JAX stack to one consistent version across ALL installs (build-time
# wheels here AND the editable repo installs in setup_python_envs.sh). Without
# this, installing crazyflow/flax later can upgrade jax/jaxlib past the CUDA
# plugin version, which breaks GPU linear algebra at runtime with:
#   "No FFI handler registered for cusolver_getrf_ffi on a platform CUDA".
jax==0.9.2
jaxlib==0.9.2
jax-cuda12-plugin==0.9.2
jax-cuda12-pjrt==0.9.2

# --- Simulator stack: PINNED (see docs/5-versions.md) ---------------------
# mujoco / mujoco-mjx MUST move together (mjx X.Y requires mujoco>=X.Y), and
# lsy_drone_racing requires mujoco<3.11 (gymnasium viewer breakage upstream).
# An unpinned image resolved mujoco 3.12 and then the racing setup downgraded
# mujoco under a 3.12 mjx -> "mujoco-mjx 3.12.0 requires mujoco>=3.12.0.dev0,
# but you have mujoco 3.10.0" (the mismatch students reported, 2026-09).
mujoco==3.10.0
mujoco-mjx==3.10.0
gymnasium==1.3.0
# crazyflow is installed EDITABLE from repos/crazyflow at the commit pinned in
# scripts/pins.sh (0.3.2); the race venv gets the same version from PyPI.
crazyflow==0.3.2
# Model files must load in both venvs: same SB3 in main (training) and race.
stable-baselines3==2.9.0
sb3-contrib==2.9.0
# casadi ships its own ipopt; the MPC of racing Lesson 6 runs in BOTH venvs (crazyflow
# depends on casadi, so the race venv has it too). Pinned so solver numerics match.
casadi==3.8.0
